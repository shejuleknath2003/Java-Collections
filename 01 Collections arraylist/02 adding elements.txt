using addAll(Collection c)
===========
public static void main(String[] args)
	{
		ArrayList al=new ArrayList();//collection type object
		al.add(10);
		al.add(20);
		al.add(30);
		al.add(40);
		al.add(50);
		
		ArrayList al2=new ArrayList();
		al2.add(5);
		al2.add(15);
		al2.add(150);
		al2.addAll(al);//adding collection
		System.out.println("al2 :"+al2);
//al2 :[5, 15, 150, 10, 20, 30, 40, 50]
		
	}
