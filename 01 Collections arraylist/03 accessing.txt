package Collectionsbatch1.accessing;

import java.util.ArrayList;

public class Demo 
{
	//accessing elements using for loop
	public static void main(String[] args)
	{
		ArrayList al=new ArrayList();//collection type object
		al.add(10);
		al.add(20);
		al.add(30);
		al.add(40);
		al.add(50);
	
		for(int i=0;i<al.size();i++)
		{
			System.out.println(al.get(i));
		}
	}
}

package Collectionsbatch1.accessing;

import java.util.ArrayList;
import java.util.Iterator;

public class DemoIterator 
{
	public static void main(String[] args) 
	{
		ArrayList<Integer> al=new ArrayList();//collection type object
		al.add(10);
		al.add(20);
		al.add(30);
		al.add(40);
		al.add(50);
		
		
		Iterator<Integer> i=al.iterator();//created iterator type object
		while(i.hasNext())
		{
			System.out.println(i.next());
		}
		System.out.println(i.next());//NoSuchElementElementException
		
	}

}

using ListIterator and traversing in forward direction :

package Collectionsbatch1.accessing;

import java.util.ArrayList;
import java.util.Iterator;
import java.util.ListIterator;

public class DemoListIterator 
{

	public static void main(String[] args) 
	{
		ArrayList<Integer> al=new ArrayList();//collection type object

		al.add(10);
		al.add(20);
		al.add(30);
		al.add(40);
		al.add(50);//4
//creates ListIterator and cursor will be pointing to just before first element
		ListIterator<Integer> li=al.listIterator();
		while(li.hasNext())
		{
			System.out.println(li.next());
		}
	}
}

package Collectionsbatch1.accessing;

import java.util.ArrayList;
import java.util.Iterator;
import java.util.ListIterator;

public class DemoListIterator 
{

	public static void main(String[] args) 
	{
		ArrayList<Integer> al=new ArrayList();//collection type object

		al.add(10);
		al.add(20);
		al.add(30);
		al.add(40);
		al.add(50);//4
		//creates ListIterator object and cursor will be pointing to just after last index
		ListIterator<Integer> li=al.listIterator(al.size());
		while(li.hasPrevious())
		{
			System.out.println(li.previous());
		}
	}
}
